       count(*) FILTER (WHERE case_id IS NULL) AS case_id_null,
       count(*) FILTER (WHERE deleted_at IS NULL) AS canli
  FROM case_documents GROUP BY 1 ORDER BY 2 DESC;
-- LINKED | 1935 | 0 | 1932   ← TEK SATIR: UNLINKED ve TEST hiç yok
SELECT min(uploaded_at)::date, max(uploaded_at)::date, count(*) FROM case_documents;

prod'da da fiilen hiç iş görmemiş. **Deploy öncesi ek kontrol gerekmiyor.**
